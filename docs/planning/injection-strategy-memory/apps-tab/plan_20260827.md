```swift
    public var rungs: [InjectionRung] { ... }
    public var health: AppsTabHealth { ... }

// The injected snapshot entry: what the wiring read off the real store + LaunchServices.
public struct AppStrategyEntry: Sendable, Equatable {
    public var bundleID: String
    public var displayName: String
    public var strategy: InjectionStrategy
}

// One row the Table renders — identity is the bundleID.
public struct AppsRow: Sendable, Equatable, Identifiable {
    public let bundleID: String
    public let displayName: String
    public let health: AppsTabHealth        // override wins over learned
    public let isOverridden: Bool           // overrideRungs != nil — shown distinctly
    public var id: String { bundleID }
}

public struct AppsTabState: Sendable, Equatable {
    public var rows: [AppsRow]              // derived from strategies, sorted by displayName
    public var isLoaded: Bool               // false until the snapshot lands
    public var saveError: String?           // the DictionarySettingsPage saveError surface
    public var strategies: [String: InjectionStrategy]   // the truth the ladder shares
}
```

Closed action set (house pattern — exhaustive switch, no time-based transitions):

```swift
public enum AppsTabAction: Sendable, Equatable {
    case snapshotLoaded([AppStrategyEntry])   // injected read; derives rows (names arrive here)
    case overrideSet(bundleID: String, method: AppsTabMethod)  // pins the rungs, marks overridden
    case overrideCleared(bundleID: String)    // back to learned
    case resetLearned                          // "reset what Vocca learned" (spec's decision)
    case saveSucceeded                         // write-through landed; clears saveError
    case saveFailed                            // write-through failed; surfaces saveError
}
```

Reducer rules (every one gets a test):

1. `snapshotLoaded` replaces `strategies`, sets `isLoaded = true`, and derives `rows`: for each
   entry, `health = effectiveFirstRung(strategy)` (override's first rung when `overrideRungs !=
   nil`, else the learned order's first rung), `isOverridden = (overrideRungs != nil)`. Rows sort
   by `displayName` (stable tiebreak: bundleID).
2. `overrideSet` sets `strategies[bundleID].overrideRungs = method.rungs`, re-derives that row's
   health/isOverridden, clears `saveError`. The row shows the **override's** health, never the
   learned one (S2). An unknown bundleID is created as a row (the override IS a strategy).
3. `overrideCleared` sets `overrideRungs = nil`, re-derives health from the learned order,
   clears `saveError`. Clearing an app that has no learned rungs either leaves its row at the
   seeded/default baseline (never-empty).
4. `resetLearned` rebuilds **every** app's strategy to `InjectionStrategy.seededBaseline(bundleID:)`
   and **re-applies the preserved override** if one exists (spec decision: overrides survive
   reset). Demotions, learned allowlist and re-probe windows are all gone. `saveError` clears.
5. `saveSucceeded` clears `saveError`; `saveFailed` sets it (the payload is the wiring's wording
   or a fixed constant — the reducer only carries a flag, copy renders the sentence, the
   `DictionarySettingsPage` rule).
6. **The reducer holds no freeze logic.** The freeze is `memory-order`'s; this reducer writes the
   override that freezes, and test 2 pins the freeze over the shipped decisions.
7. No time-based transition exists (the never-auto-dismiss rule).

## Edge cases

- **Empty store:** `isLoaded && rows.isEmpty` → the empty-state copy, not a blank table
  (`DictionarySettingsPage`'s `isLoaded` guard).
- **App with no strategy yet:** not a row (the tab is "what Vocca learned / pinned"); an
  override on an unknown app creates its row. An app whose name fails to resolve shows the
  bundleID (never a nameless row).
- **Overridden vs learned rows:** `isOverridden` drives both the health (override's) and the
  distinct "pinned by you" indicator; the two never blur.
- **Reset on overrides:** overrides survive (spec Q1 decision) — a reset must not read as
  "Vocca undid what I told it".
- **Save failure:** surfaced via `saveError`, cleared on the next success (the
  `DictionarySettingsPage` rule); a store that silently fails to save is one the user rebuilds
  from scratch.
- **A `widgetFailsafe`-only effective order:** cannot occur (R2: never in a strategy); the
  health mapping still defines it (→ `manual only`) so the table is total.
